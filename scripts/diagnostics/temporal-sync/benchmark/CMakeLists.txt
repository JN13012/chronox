cmake_minimum_required(VERSION 3.24)
project(TemporalBenchmark LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(REFERENCE "${ROOT}/build/temporal-reference")
add_library(bench_sqlite STATIC "${ROOT}/build/_deps/sqlite3-src/sqlite3.c")
target_include_directories(bench_sqlite PUBLIC "${ROOT}/build/_deps/sqlite3-src")
target_compile_definitions(bench_sqlite PRIVATE SQLITE_DQS=0 SQLITE_THREADSAFE=1)
target_compile_options(bench_sqlite PRIVATE /W0)
foreach(flavor baseline measured)
    if(flavor STREQUAL baseline)
        set(SOURCE "${REFERENCE}/src")
    else()
        set(SOURCE "${ROOT}/src")
    endif()
    add_executable(temporal_bench_${flavor} temporal_benchmark.cpp
        "${SOURCE}/core/calendar_period.cpp" "${SOURCE}/core/session_tracker.cpp"
        "${SOURCE}/core/utc_timestamp_clock.cpp" "${SOURCE}/storage/session_database.cpp")
    target_include_directories(temporal_bench_${flavor} PRIVATE "${SOURCE}/core" "${SOURCE}/storage")
    target_compile_features(temporal_bench_${flavor} PRIVATE cxx_std_20)
    target_compile_definitions(temporal_bench_${flavor} PRIVATE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(temporal_bench_${flavor} PRIVATE bench_sqlite Psapi Mincore)
endforeach()
target_compile_definitions(temporal_bench_baseline PRIVATE CHRONOX_TEMPORAL_BASELINE)
target_sources(temporal_bench_measured PRIVATE "${ROOT}/src/core/temporal_engine.cpp"
    "${ROOT}/src/core/temporal_diagnostics.cpp" "${ROOT}/src/windows/windows_time_source.cpp")
target_include_directories(temporal_bench_measured PRIVATE "${ROOT}/src/windows")
